   180→   180→  const texts = useRef<Array<HTMLDivElement | null>>([]);
   181→   181→  const ticks = useRef<Array<HTMLSpanElement | null>>([]);
   182→   182→  /* client-only mount (behind the experience's WebGL gate) — the media
   183→   183→     query is safe to read at first render */
   184→   184→  const [reduced] = useState(() => prefersReducedMotion());
   185→   185→
   186→   186→  useEffect(() => {
   187→   187→    if (reduced || !root.current || !sticky.current) return;
   188→   188→    const l0 = layers.current[0];
   189→   189→    const l1 = layers.current[1];
   190→   190→    const l2 = layers.current[2];
   191→   191→    const t0 = texts.current[0];
   192→   192→    const t1 = texts.current[1];
   193→   193→    const t2 = texts.current[2];
   194→   194→    if (!l0 || !l1 || !l2 || !t0 || !t1 || !t2) return;
   195→   195→
   196→   196→    const ctx = gsap.context(() => {
   197→   197→      /* the two dissolves sit at these beats of the pinned travel */
   198→   198→      const B1 = 0.335;
   199→   199→      const B2 = 0.67;
   200→   200→
   201→   201→      /* initial states — everything waits in the dark */
   202→   202→      gsap.set([l1, l2], { autoAlpha: 0 });
   203→   203→      gsap.set([t0, t1, t2], { autoAlpha: 0, y: 48 });
   204→   204→      gsap.set(l0, { scale: 1.16, autoAlpha: 0 });
   205→   205→      gsap.set(l1, { scale: 1.18, xPercent: -2.5 });
   206→   206→      gsap.set(l2, { scale: 1.16, xPercent: 2 });
   207→   207→
   208→   208→      /* entry dissolve — the first terrain rises out of the chapter dark
   209→   209→         while the title card above scrolls away */
   210→   210→      gsap.to(l0, {
   211→   211→        autoAlpha: 1,
   212→   212→        ease: "none",
   213→   213→        scrollTrigger: {
   214→   214→          trigger: root.current!,
   215→   215→          start: "top bottom",
   216→   216→          end: "top top",
   217→   217→          scrub: 0.5,
   218→   218→        },
   219→   219→      });
   220→   220→
   221→   221→      let active = -1;
   222→   222→      const tl = gsap.timeline({
   223→   223→        defaults: { ease: "none" },
   224→   224→        scrollTrigger: {
   225→   225→          trigger: root.current!,
   226→   226→          start: "top top",
   227→   227→          end: "bottom bottom",
   228→   228→          scrub: 0.7,
   229→   229→          onUpdate: (self) => {
   230→   230→            const p = self.progress;
   231→   231→            const i = p < 0.34 ? 0 : p < 0.675 ? 1 : 2;
   232→   232→            if (i === active) return;
   233→   233→            active = i;
   234→   234→            const accent = SECTORS[i].theme.accent;
   235→   235→            sticky.current!.style.setProperty("--accent", accent);
   236→   236→            texts.current.forEach((el, k) => {
   237→   237→              if (!el) return;
   238→   238→              el.inert = k !== i;
   239→   239→              el.style.pointerEvents = k === i ? "auto" : "none";
   240→   240→            });
   241→   241→            ticks.current.forEach((el, k) => {
   242→   242→              if (!el) return;
   243→   243→              el.style.width = k === i ? "26px" : "14px";
   244→   244→              el.style.backgroundColor =
   245→   245→                k === i ? accent : "rgba(255,255,255,0.28)";
   246→   246→            });
   247→   247→          },
   248→   248→        },
   249→   249→      });
   250→   250→
   251→   251→      /* — terrain 01 · the fault line — */
   252→   252→      tl.to(l0, { scale: 1.03, duration: 0.385 }, 0);
   253→   253→      tl.to(t0, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.02);
   254→   254→      tl.to(t0, { autoAlpha: 0, y: -48, duration: 0.07, ease: "power2.in" }, 0.235);
   255→   255→
   256→   256→      /* — dissolve — */
   257→   257→      tl.to(l1, { autoAlpha: 1, duration: 0.1 }, B1 - 0.055);
   258→   258→      tl.to(l1, { scale: 1.045, xPercent: 0, duration: 0.37 }, 0.3);
   259→   259→      tl.set(l0, { visibility: "hidden" }, 0.4);
   260→   260→
   261→   261→      /* — terrain 02 · the delta — */
   262→   262→      tl.to(t1, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.385);
   263→   263→      tl.to(t1, { autoAlpha: 0, y: -48, duration: 0.07, ease: "power2.in" }, 0.575);
   264→   264→
   265→   265→      /* — dissolve — */
   266→   266→      tl.to(l2, { autoAlpha: 1, duration: 0.1 }, B2 - 0.055);
   267→   267→      tl.to(l2, { scale: 1.035, xPercent: 0, duration: 0.355 }, 0.645);
   268→   268→      tl.set(l1, { visibility: "hidden" }, 0.73);
   269→   269→
   270→   270→      /* — terrain 03 · the river — stays on stage; the next chapter
   271→   271→           melts up over it through its own gradient seam */
   272→   272→      tl.to(t2, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.72);
   273→   273→    }, root.current);
   274→   274→
   275→   275→    return () => ctx.revert();
   276→   276→  }, [reduced]);
   277→   277→
   278→   278→  if (reduced) {
   279→   279→    return (
   280→   280→      <div className="relative">
   281→   281→        {SECTORS.map((s, i) => (
   282→   282→          <StaticSector key={s.id} sector={s} first={i === 0} />
   283→   283→        ))}
   284→   284→      </div>
   285→   285→    );
   286→   286→  }
   287→   287→
   288→   288→  return (
   289→   289→    <div
   290→   290→      ref={root}
   291→   291→      data-frame
   292→   292→      role="region"
   293→   293→      aria-label="The changing systems"
   294→   294→      className="relative z-[1]"
   295→   295→      style={{ height: "454svh" }}
   296→   296→    >
   297→   297→      <div
   298→   298→        ref={sticky}
   299→   299→        className="sticky top-0 h-svh overflow-hidden bg-[#04070d]"
   300→   300→        style={{ ["--accent" as string]: SECTORS[0].theme.accent }}
   301→   301→      >
   302→   302→        {/* the terrains — stacked, dissolved one into the next */}
   303→   303→        {SECTORS.map((s, i) => (
   304→   304→          <div
   305→   305→            key={s.id}
   306→   306→            ref={(el) => {
   307→   307→              layers.current[i] = el;
   308→   308→            }}
   309→   309→            className="absolute inset-0 will-change-transform"
   310→   310→            style={{ zIndex: 10 + i * 10 }}
   311→   311→          >
   312→   312→            <Image
   313→   313→              src={s.image}
   314→   314→              alt={s.imageAlt}
   315→   315→              fill
   316→   316→              sizes="100vw"
   317→   317→              quality={82}
   318→   318→              priority={i === 0}
   319→   319→              loading="eager"
   320→   320→              className="object-cover"
   321→   321→            />
   322→   322→            <div
   323→   323→              aria-hidden
   324→   324→              className={`absolute inset-0 ${
   325→   325→                s.align === "left"
   326→   326→                  ? "bg-gradient-to-r from-[#04070d]/92 via-[#04070d]/38 to-transparent"
   327→   327→                  : "bg-gradient-to-l from-[#04070d]/92 via-[#04070d]/38 to-transparent"
   328→   328→              }`}
   329→   329→            />
   330→   330→            <div
   331→   331→              aria-hidden
   332→   332→              className="absolute inset-0 bg-gradient-to-t from-[#04070d]/60 via-transparent to-[#04070d]/40"

   336→   336→
   337→   337→        {/* the grade — the same vignette and grain finish as the Earth stage,

   339→   339→        <div

   398→   398→/* the compare slider — optical vs radar, draggable                     */
   399→   399→/* ------------------------------------------------------------------ */
   400→   400→
   401→   401→function CompareSlider() {
   402→   402→  const root = useRef<HTMLDivElement>(null);
   403→   403→  const clip = useRef<HTMLDivElement>(null);
   404→   404→  const divider = useRef<HTMLDivElement>(null);
   405→   405→  const readout = useRef<HTMLSpanElement>(null);
   406→   406→  const posRef = useRef(35);
   407→   407→  const dragging = useRef(false);
   408→   408→
   409→   409→  const apply = useCallback((v: number) => {
   410→   410→    posRef.current = v;
   411→   411→    if (divider.current) divider.current.style.left = `${v}%`;
   412→   412→    if (clip.current) clip.current.style.clipPath = `inset(0 0 0 ${v}%)`;
   413→   413→    if (readout.current)
   414→   414→      readout.current.textContent = `RADAR ${Math.round(v)}%`;
   415→   415→    if (root.current)
   416→   416→      root.current.setAttribute("aria-valuenow", String(Math.round(v)));
   417→   417→  }, []);
   418→   418→
   419→   419→  /* the radar intro — the frame wipes in, then the divider makes one
   420→   420→     scanning pass across the scene before settling, the way the engine
   421→   421→     itself sweeps a scene */
   422→   422→  useEffect(() => {
   423→   423→    if (prefersReducedMotion()) return;
   424→   424→    const el = root.current;
   425→   425→    if (!el) return;
   426→   426→    const ctx = gsap.context(() => {
   427→   427→      gsap.fromTo(
   428→   428→        el,
   429→   429→        { clipPath: "inset(0 100% 0 0)" },
   430→   430→        {
   431→   431→          clipPath: "inset(0 0% 0 0)",
   432→   432→          duration: 1.4,
   433→   433→          ease: "power3.inOut",
   434→   434→          scrollTrigger: { trigger: el, start: "top 80%", once: true },
   435→   435→          onComplete: () => {
   436→   436→            const o = { v: 97 };
   437→   437→            gsap.to(o, {
   438→   438→              v: 35,
   439→   439→              duration: 2.1,
   440→   440→              ease: "power2.inOut",
   441→   441→              onUpdate: () => apply(o.v),
   442→   442→            });
   443→   443→          },
   444→   444→        },
   445→   445→      );
   446→   446→    });
   447→   447→    return () => ctx.revert();
   448→   448→  }, [apply]);
   449→   449→
   450→   450→  const setFromClientX = useCallback(
   451→   451→    (clientX: number) => {
   452→   452→      const el = root.current;
   453→   453→      if (!el) return;
   454→   454→      const r = el.getBoundingClientRect();
   455→   455→      apply(clamp(((clientX - r.left) / r.width) * 100, 2, 98));
   456→   456→    },
   457→   457→    [apply],
   458→   458→  );
   459→   459→
   460→   460→  const onPointerDown = (e: ReactPointerEvent<HTMLDivElement>) => {
   461→   461→    dragging.current = true;
   462→   462→    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);

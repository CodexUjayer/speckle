   176→function SystemsSequence() {
   177→  const root = useRef<HTMLDivElement>(null);
   178→  const sticky = useRef<HTMLDivElement>(null);
   179→  const layers = useRef<Array<HTMLDivElement | null>>([]);
   180→  const texts = useRef<Array<HTMLDivElement | null>>([]);
   181→  const ticks = useRef<Array<HTMLSpanElement | null>>([]);
   182→  /* client-only mount (behind the experience's WebGL gate) — the media
   183→     query is safe to read at first render */
   184→  const [reduced] = useState(() => prefersReducedMotion());
   185→
   186→  useEffect(() => {
   187→    if (reduced || !root.current || !sticky.current) return;
   188→    const l0 = layers.current[0];
   189→    const l1 = layers.current[1];
   190→    const l2 = layers.current[2];
   191→    const t0 = texts.current[0];
   192→    const t1 = texts.current[1];
   193→    const t2 = texts.current[2];
   194→    if (!l0 || !l1 || !l2 || !t0 || !t1 || !t2) return;
   195→
   196→    const ctx = gsap.context(() => {
   197→      /* the two dissolves sit at these beats of the pinned travel */
   198→      const B1 = 0.335;
   199→      const B2 = 0.67;
   200→
   201→      /* initial states — everything waits in the dark */
   202→      gsap.set([l1, l2], { autoAlpha: 0 });
   203→      gsap.set([t0, t1, t2], { autoAlpha: 0, y: 48 });
   204→      gsap.set(l0, { scale: 1.16, autoAlpha: 0 });
   205→      gsap.set(l1, { scale: 1.18, xPercent: -2.5 });
   206→      gsap.set(l2, { scale: 1.16, xPercent: 2 });
   207→
   208→      /* entry dissolve — the first terrain rises out of the chapter dark
   209→         while the title card above scrolls away */
   210→      gsap.to(l0, {
   211→        autoAlpha: 1,
   212→        ease: "none",
   213→        scrollTrigger: {
   214→          trigger: root.current!,
   215→          start: "top bottom",
   216→          end: "top top",
   217→          scrub: 0.5,
   218→        },
   219→      });
   220→
   221→      let active = -1;
   222→      const tl = gsap.timeline({
   223→        defaults: { ease: "none" },
   224→        scrollTrigger: {
   225→          trigger: root.current!,
   226→          start: "top top",
   227→          end: "bottom bottom",
   228→          scrub: 0.7,
   229→          onUpdate: (self) => {
   230→            const p = self.progress;
   231→            const i = p < 0.34 ? 0 : p < 0.675 ? 1 : 2;
   232→            if (i === active) return;
   233→            active = i;
   234→            const accent = SECTORS[i].theme.accent;
   235→            sticky.current!.style.setProperty("--accent", accent);
   236→            texts.current.forEach((el, k) => {
   237→              if (!el) return;
   238→              el.inert = k !== i;
   239→              el.style.pointerEvents = k === i ? "auto" : "none";
   240→            });
   241→            ticks.current.forEach((el, k) => {
   242→              if (!el) return;
   243→              el.style.width = k === i ? "26px" : "14px";
   244→              el.style.backgroundColor =
   245→                k === i ? accent : "rgba(255,255,255,0.28)";
   246→            });
   247→          },
   248→        },
   249→      });
   250→
   251→      /* — terrain 01 · the fault line — */
   252→      tl.to(l0, { scale: 1.03, duration: 0.385 }, 0);
   253→      tl.to(t0, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.02);
   254→      tl.to(t0, { autoAlpha: 0, y: -48, duration: 0.07, ease: "power2.in" }, 0.235);
   255→
   256→      /* — dissolve — */
   257→      tl.to(l1, { autoAlpha: 1, duration: 0.1 }, B1 - 0.055);
   258→      tl.to(l1, { scale: 1.045, xPercent: 0, duration: 0.37 }, 0.3);
   259→      tl.set(l0, { visibility: "hidden" }, 0.4);
   260→
   261→      /* — terrain 02 · the delta — */
   262→      tl.to(t1, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.385);
   263→      tl.to(t1, { autoAlpha: 0, y: -48, duration: 0.07, ease: "power2.in" }, 0.575);
   264→
   265→      /* — dissolve — */
   266→      tl.to(l2, { autoAlpha: 1, duration: 0.1 }, B2 - 0.055);
   267→      tl.to(l2, { scale: 1.035, xPercent: 0, duration: 0.355 }, 0.645);
   268→      tl.set(l1, { visibility: "hidden" }, 0.73);
   269→
   270→      /* — terrain 03 · the river — stays on stage; the next chapter
   271→           melts up over it through its own gradient seam */
   272→      tl.to(t2, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.72);
   273→    }, root.current);
   274→
   275→    return () => ctx.revert();
   276→  }, [reduced]);
   277→
   278→  if (reduced) {
   279→    return (
   280→      <div className="relative">
   281→        {SECTORS.map((s, i) => (
   282→          <StaticSector key={s.id} sector={s} first={i === 0} />
   283→        ))}
   284→      </div>
   285→    );
   286→  }
   287→
   288→  return (
   289→    <div
   290→      ref={root}
   291→      data-frame
   292→      role="region"
   293→      aria-label="The changing systems"
   294→      className="relative z-[1]"
   295→      style={{ height: "454svh" }}
   296→    >
   297→      <div
   298→        ref={sticky}
   299→        className="sticky top-0 h-svh overflow-hidden bg-[#04070d]"
   300→        style={{ ["--accent" as string]: SECTORS[0].theme.accent }}
   301→      >
   302→        {/* the terrains — stacked, dissolved one into the next */}
   303→        {SECTORS.map((s, i) => (
   304→          <div
   305→            key={s.id}
   306→            ref={(el) => {
   307→              layers.current[i] = el;
   308→            }}
   309→            className="absolute inset-0 will-change-transform"
   310→            style={{ zIndex: 10 + i * 10 }}
   311→          >
   312→            <Image
   313→              src={s.image}
   314→              alt={s.imageAlt}
   315→              fill
   316→              sizes="100vw"
   317→              quality={82}
   318→              priority={i === 0}
   319→              loading="eager"
   320→              className="object-cover"
   321→            />
   322→            <div
   323→              aria-hidden
   324→              className={`absolute inset-0 ${
   325→                s.align === "left"
   326→                  ? "bg-gradient-to-r from-[#04070d]/92 via-[#04070d]/38 to-transparent"
   327→                  : "bg-gradient-to-l from-[#04070d]/92 via-[#04070d]/38 to-transparent"
   328→              }`}
   329→            />
   330→            <div
   331→              aria-hidden
   332→              className="absolute inset-0 bg-gradient-to-t from-[#04070d]/60 via-transparent to-[#04070d]/40"
   333→            />
   334→          </div>
   335→        ))}
   336→
   337→        {/* the grade — the same vignette and grain finish as the Earth stage,
   338→            so the 2D chapters feel cut from the same film */}
   339→        <div
   340→          aria-hidden
   341→          className="pointer-events-none absolute inset-0 z-40"
   342→          style={{
   343→            background:
   344→              "radial-gradient(ellipse 120% 90% at 50% 45%, rgba(4,7,13,0) 55%, rgba(4,7,13,0.3) 84%, rgba(4,7,13,0.55) 100%)",
   345→          }}
   346→        />
   347→        <div
   348→          aria-hidden
   349→          className="grain pointer-events-none absolute inset-0 z-40 opacity-70"
   350→        />
   351→
   352→        {/* the words — one subject per terrain. Only the system on stage
   353→            takes pointer events and focus. */}
   354→        {SECTORS.map((s, i) => (
   355→          <div
   356→            key={s.id}
   357→            ref={(el) => {
   358→              texts.current[i] = el;
   359→            }}
   360→            inert={i !== 0}
   361→            className="absolute inset-0 z-50 overflow-y-auto overscroll-contain"
   362→            style={{ pointerEvents: i === 0 ? "auto" : "none" }}
   363→          >
   364→            <div className="flex min-h-full items-center py-8">
   365→              <div className="mx-auto w-full max-w-7xl px-5 sm:px-8">
   366→                <SystemText sector={s} />
   367→              </div>
   368→            </div>
   369→          </div>
   370→        ))}
   371→
   372→        {/* the sequence rail — three quiet ticks */}
   373→        <div
   374→          aria-hidden
   375→          className="absolute right-2.5 top-1/2 z-50 flex -translate-y-1/2 flex-col items-end gap-2.5 sm:right-7"
   376→        >
   377→          {SECTORS.map((s, i) => (
   378→            <span
   379→              key={s.id}
   380→              ref={(el) => {
   381→                ticks.current[i] = el;
   382→              }}
   383→              className="block h-px transition-all duration-500"
   384→              style={{
   385→                width: i === 0 ? 26 : 14,
   386→                backgroundColor:
   387→                  i === 0 ? s.theme.accent : "rgba(255,255,255,0.28)",
   388→              }}
   389→            />
   390→          ))}
   391→        </div>
   392→      </div>
   393→    </div>
   394→  );
   395→}

   397→/* ------------------------------------------------------------------ */
   398→/* the compare slider — optical vs radar, draggable                     */
   399→/* ------------------------------------------------------------------ */
   400→
   401→function CompareSlider() {
   402→  const root = useRef<HTMLDivElement>(null);
   403→  const clip = useRef<HTMLDivElement>(null);
   404→  const divider = useRef<HTMLDivElement>(null);
   405→  const readout = useRef<HTMLSpanElement>(null);
   406→  const posRef = useRef(35);
   407→  const dragging = useRef(false);
   408→
   409→  const apply = useCallback((v: number) => {
   410→    posRef.current = v;
   411→    if (divider.current) divider.current.style.left = `${v}%`;
   412→    if (clip.current) clip.current.style.clipPath = `inset(0 0 0 ${v}%)`;
   413→    if (readout.current)
   414→      readout.current.textContent = `RADAR ${Math.round(v)}%`;
   415→    if (root.current)
   416→      root.current.setAttribute("aria-valuenow", String(Math.round(v)));
   417→  }, []);
   418→
   419→  /* the radar intro — the frame wipes in, then the divider makes one
   420→     scanning pass across the scene before settling, the way the engine
   421→     itself sweeps a scene */
   422→  useEffect(() => {
   423→    if (prefersReducedMotion()) return;
   424→    const el = root.current;
   425→    if (!el) return;
   426→    const ctx = gsap.context(() => {
   427→      gsap.fromTo(
   428→        el,
   429→        { clipPath: "inset(0 100% 0 0)" },
   430→        {
   431→          clipPath: "inset(0 0% 0 0)",
   432→          duration: 1.4,
   433→          ease: "power3.inOut",
   434→          scrollTrigger: { trigger: el, start: "top 80%", once: true },
   435→          onComplete: () => {
   436→            const o = { v: 97 };
   437→            gsap.to(o, {
   438→              v: 35,
   439→              duration: 2.1,
   440→              ease: "power2.inOut",
   441→              onUpdate: () => apply(o.v),
   442→            });
   443→          },
   444→        },
   445→      );
   446→    });
   447→    return () => ctx.revert();
   448→  }, [apply]);
   449→
   450→  const setFromClientX = useCallback(
   451→    (clientX: number) => {
   452→      const el = root.current;
   453→      if (!el) return;
   454→      const r = el.getBoundingClientRect();
   455→      apply(clamp(((clientX - r.left) / r.width) * 100, 2, 98));
   456→    },
   457→    [apply],
   458→  );
   459→
   460→  const onPointerDown = (e: ReactPointerEvent<HTMLDivElement>) => {
   461→    dragging.current = true;
   462→    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
   463→    setFromClientX(e.clientX);
   464→  };
   465→  const onPointerMove = (e: ReactPointerEvent<HTMLDivElement>) => {
   466→    if (dragging.current) setFromClientX(e.clientX);
   467→  };
   468→  const endDrag = () => {
   469→    dragging.current = false;
   470→  };
   471→
   472→  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
   473→    const step = e.shiftKey ? 10 : 4;
   474→    if (e.key === "ArrowLeft") {
   475→      apply(clamp(posRef.current - step, 2, 98));
   476→      e.preventDefault();
   477→    } else if (e.key === "ArrowRight") {
   478→      apply(clamp(posRef.current + step, 2, 98));
   479→      e.preventDefault();
   480→    } else if (e.key === "Home") {
   481→      apply(2);
   482→      e.preventDefault();
   483→    } else if (e.key === "End") {
   484→      apply(98);
   485→      e.preventDefault();
   486→    }
   487→  };
   488→
   489→  return (
   490→    <div>
   491→      <div
   492→        ref={root}
   493→        role="slider"
   494→        tabIndex={0}
   495→        aria-valuemin={0}
   496→        aria-valuemax={100}
   497→        aria-valuenow={35}
   498→        aria-label="Comparison between optical and radar imagery"
   499→        aria-valuetext="35% radar"
   500→        onPointerDown={onPointerDown}
   501→        onPointerMove={onPointerMove}
   502→        onPointerUp={endDrag}
   503→        onPointerCancel={endDrag}
   504→        onKeyDown={onKeyDown}
   505→        className="relative aspect-[16/9] w-full cursor-ew-resize touch-pan-y select-none overflow-hidden rounded-md border border-white/15 bg-[#0a0f18] shadow-[0_30px_90px_rgba(0,0,0,0.5)] focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-4 focus-visible:outline-[var(--accent)]"
   506→      >
   507→        {/* optical (base) */}
   508→        <div
   509→          className="absolute inset-0 bg-cover bg-center"
   510→          style={{ backgroundImage: "url(/landscapes/earthquake.webp)" }}
   511→          aria-hidden
   512→        />
   513→        {/* radar (clipped) */}
   514→        <div
   515→          ref={clip}
   516→          className="absolute inset-0 bg-cover bg-center"
   517→          style={{
   518→            backgroundImage: "url(/landscapes/sar.webp)",
   519→            clipPath: "inset(0 0 0 35%)",
   520→          }}
   521→          aria-hidden
   522→        />
   523→
   524→        <span className="pointer-events-none absolute left-3 top-3 rounded-sm bg-[#04070d]/78 px-2.5 py-1.5 text-[10px] font-medium uppercase tracking-[0.2em] text-white/80 backdrop-blur-sm">
   525→          {EVIDENCE.compareLabels.left}
   526→        </span>
   527→        <span className="pointer-events-none absolute right-3 top-3 rounded-sm bg-[#04070d]/78 px-2.5 py-1.5 text-[10px] font-medium uppercase tracking-[0.2em] text-white/80 backdrop-blur-sm">
   528→          {EVIDENCE.compareLabels.right}
   529→        </span>
   530→
   531→        <div
   532→          ref={divider}
   533→          className="pointer-events-none absolute inset-y-0 w-px bg-white/90"
   534→          style={{ left: "35%" }}
   535→        >
   536→          <span className="absolute left-1/2 top-1/2 flex h-10 w-10 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-full border border-white/75 bg-[#04070d]/75 text-white backdrop-blur-sm">
   537→            <MoveHorizontal className="h-4 w-4" aria-hidden />
   538→          </span>
   539→        </div>
   540→

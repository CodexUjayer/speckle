   181→function SystemsSequence() {
   182→  const root = useRef<HTMLDivElement>(null);
   183→  const sticky = useRef<HTMLDivElement>(null);
   184→  const layers = useRef<Array<HTMLDivElement | null>>([]);
   185→  const texts = useRef<Array<HTMLDivElement | null>>([]);
   186→  const ticks = useRef<Array<HTMLSpanElement | null>>([]);
   187→  /* client-only mount (behind the experience's WebGL gate) — the media
   188→     query is safe to read at first render */
   189→  const [reduced] = useState(() => prefersReducedMotion());
   190→
   191→  useEffect(() => {
   192→    if (reduced || !root.current || !sticky.current) return;
   193→    const l0 = layers.current[0];
   194→    const l1 = layers.current[1];
   195→    const l2 = layers.current[2];
   196→    const t0 = texts.current[0];
   197→    const t1 = texts.current[1];
   198→    const t2 = texts.current[2];
   199→    if (!l0 || !l1 || !l2 || !t0 || !t1 || !t2) return;
   200→
   201→    const ctx = gsap.context(() => {
   202→      /* the two dissolves sit at these beats of the pinned travel */
   203→      const B1 = 0.335;
   204→      const B2 = 0.67;
   205→
   206→      /* initial states — everything waits in the dark */
   207→      gsap.set([l1, l2], { autoAlpha: 0 });
   208→      gsap.set([t0, t1, t2], { autoAlpha: 0, y: 48 });
   209→      gsap.set(l0, { scale: 1.16, autoAlpha: 0 });
   210→      gsap.set(l1, { scale: 1.18, xPercent: -2.5 });
   211→      gsap.set(l2, { scale: 1.16, xPercent: 2 });
   212→
   213→      /* entry dissolve — the first terrain rises out of the chapter dark
   214→         while the title card above scrolls away */
   215→      gsap.to(l0, {
   216→        autoAlpha: 1,
   217→        ease: "none",
   218→        scrollTrigger: {
   219→          trigger: root.current!,
   220→          start: "top bottom",
   221→          end: "top top",
   222→          scrub: 0.5,
   223→        },
   224→      });
   225→
   226→      let active = -1;
   227→      const tl = gsap.timeline({
   228→        defaults: { ease: "none" },
   229→        scrollTrigger: {
   230→          trigger: root.current!,
   231→          start: "top top",
   232→          end: "bottom bottom",
   233→          scrub: 0.7,
   234→          onUpdate: (self) => {
   235→            const p = self.progress;
   236→            const i = p < 0.34 ? 0 : p < 0.675 ? 1 : 2;
   237→            if (i === active) return;
   238→            active = i;
   239→            const accent = SECTORS[i].theme.accent;
   240→            sticky.current!.style.setProperty("--accent", accent);
   241→            texts.current.forEach((el, k) => {
   242→              if (!el) return;
   243→              el.inert = k !== i;
   244→              el.style.pointerEvents = k === i ? "auto" : "none";
   245→            });
   246→            ticks.current.forEach((el, k) => {
   247→              if (!el) return;
   248→              el.style.width = k === i ? "26px" : "14px";
   249→              el.style.backgroundColor =
   250→                k === i ? accent : "rgba(255,255,255,0.28)";
   251→            });
   252→          },
   253→        },
   254→      });
   255→
   256→      /* — terrain 01 · the fault line — */
   257→      tl.to(l0, { scale: 1.03, duration: 0.385 }, 0);
   258→      tl.to(t0, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.02);
   259→      tl.to(t0, { autoAlpha: 0, y: -48, duration: 0.07, ease: "power2.in" }, 0.235);
   260→
   261→      /* — dissolve — */
   262→      tl.to(l1, { autoAlpha: 1, duration: 0.1 }, B1 - 0.055);
   263→      tl.to(l1, { scale: 1.045, xPercent: 0, duration: 0.37 }, 0.3);
   264→      tl.set(l0, { visibility: "hidden" }, 0.4);
   265→
   266→      /* — terrain 02 · the delta — */
   267→      tl.to(t1, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.385);
   268→      tl.to(t1, { autoAlpha: 0, y: -48, duration: 0.07, ease: "power2.in" }, 0.575);
   269→
   270→      /* — dissolve — */
   271→      tl.to(l2, { autoAlpha: 1, duration: 0.1 }, B2 - 0.055);
   272→      tl.to(l2, { scale: 1.035, xPercent: 0, duration: 0.355 }, 0.645);
   273→      tl.set(l1, { visibility: "hidden" }, 0.73);
   274→
   275→      /* — terrain 03 · the river — stays on stage; the next chapter
   276→           melts up over it through its own gradient seam */
   277→      tl.to(t2, { autoAlpha: 1, y: 0, duration: 0.075, ease: "power2.out" }, 0.72);
   278→    }, root.current);
   279→
   280→    return () => ctx.revert();
   281→  }, [reduced]);
   282→
   283→  if (reduced) {
   284→    return (
   285→      <div className="relative">
   286→        {SECTORS.map((s, i) => (
   287→          <StaticSector key={s.id} sector={s} first={i === 0} />
   288→        ))}
   289→      </div>
   290→    );
   291→  }
   292→
   293→  return (
   294→    <div
   295→      ref={root}
   296→      id="systems"
   297→      data-frame
   298→      role="region"
   299→      aria-label="The changing systems"
   300→      className="relative z-[1]"
   301→      style={{ height: "454svh" }}
   302→    >
   303→      <div
   304→        ref={sticky}
   305→        className="sticky top-0 h-svh overflow-hidden bg-[#04070d]"
   306→        style={{ ["--accent" as string]: SECTORS[0].theme.accent }}
   307→      >
   308→        {/* the terrains — stacked, dissolved one into the next */}
   309→        {SECTORS.map((s, i) => (
   310→          <div
   311→            key={s.id}
   312→            ref={(el) => {
   313→              layers.current[i] = el;
   314→            }}
   315→            className="absolute inset-0 will-change-transform"
   316→            style={{ zIndex: 10 + i * 10 }}
   317→          >
   318→            <Image
   319→              src={s.image}
   320→              alt={s.imageAlt}
   321→              fill
   322→              sizes="100vw"
   323→              quality={82}
   324→              priority={i === 0}
   325→              loading="eager"
   326→              className="object-cover"
   327→            />
   328→            <div
   329→              aria-hidden
   330→              className={`absolute inset-0 ${
   331→                s.align === "left"
   332→                  ? "bg-gradient-to-r from-[#04070d]/92 via-[#04070d]/38 to-transparent"
   333→                  : "bg-gradient-to-l from-[#04070d]/92 via-[#04070d]/38 to-transparent"
   334→              }`}
   335→            />
   336→            <div
   337→              aria-hidden
   338→              className="absolute inset-0 bg-gradient-to-t from-[#04070d]/60 via-transparent to-[#04070d]/40"
   339→            />
   340→          </div>
   341→        ))}
   342→
   343→        {/* the grade — the same vignette and grain finish as the Earth stage,
   344→            so the 2D chapters feel cut from the same film */}
   345→        <div
   346→          aria-hidden
   347→          className="pointer-events-none absolute inset-0 z-40"
   348→          style={{
   349→            background:
   350→              "radial-gradient(ellipse 120% 90% at 50% 45%, rgba(4,7,13,0) 55%, rgba(4,7,13,0.3) 84%, rgba(4,7,13,0.55) 100%)",
   351→          }}
   352→        />
   353→        <div
   354→          aria-hidden
   355→          className="grain pointer-events-none absolute inset-0 z-40 opacity-70"
   356→        />
   357→
   358→        {/* the words — one subject per terrain. Only the system on stage
   359→            takes pointer events and focus. */}
   360→        {SECTORS.map((s, i) => (
   361→          <div
   362→            key={s.id}
   363→            ref={(el) => {
   364→              texts.current[i] = el;
   365→            }}
   366→            inert={i !== 0}
   367→            className="absolute inset-0 z-50 overflow-y-auto overscroll-contain"
   368→            style={{ pointerEvents: i === 0 ? "auto" : "none" }}
   369→          >
   370→            <div className="flex min-h-full items-center py-8">
   371→              <div className="mx-auto w-full max-w-7xl px-5 sm:px-8">
   372→                <SystemText sector={s} />
   373→              </div>
   374→            </div>
   375→          </div>
   376→        ))}
   377→
   378→        {/* the sequence rail — three quiet ticks */}
   379→        <div
   380→          aria-hidden
   381→          className="absolute right-2.5 top-1/2 z-50 flex -translate-y-1/2 flex-col items-end gap-2.5 sm:right-7"
   382→        >
   383→          {SECTORS.map((s, i) => (
   384→            <span
   385→              key={s.id}
   386→              ref={(el) => {
   387→                ticks.current[i] = el;
   388→              }}
   389→              className="block h-px transition-all duration-500"
   390→              style={{
   391→                width: i === 0 ? 26 : 14,
   392→                backgroundColor:
   393→                  i === 0 ? s.theme.accent : "rgba(255,255,255,0.28)",
   394→              }}
   395→            />
   396→          ))}
   397→        </div>
   398→      </div>
   399→    </div>
   400→  );
   401→}

   403→/* ------------------------------------------------------------------ */
   404→/* the compare slider — optical vs radar, draggable                     */
   405→/* ------------------------------------------------------------------ */
   406→
   407→function CompareSlider() {
   408→  const root = useRef<HTMLDivElement>(null);
   409→  const clip = useRef<HTMLDivElement>(null);
   410→  const divider = useRef<HTMLDivElement>(null);
   411→  const readout = useRef<HTMLSpanElement>(null);
   412→  const posRef = useRef(35);
   413→  const dragging = useRef(false);
   414→
   415→  const apply = useCallback((v: number) => {
   416→    posRef.current = v;
   417→    if (divider.current) divider.current.style.left = `${v}%`;
   418→    if (clip.current) clip.current.style.clipPath = `inset(0 0 0 ${v}%)`;
   419→    if (readout.current)
   420→      readout.current.textContent = `RADAR ${Math.round(v)}%`;
   421→    if (root.current)
   422→      root.current.setAttribute("aria-valuenow", String(Math.round(v)));
   423→  }, []);
   424→
   425→  /* the radar intro — the frame wipes in, then the divider makes one
   426→     scanning pass across the scene before settling, the way the engine
   427→     itself sweeps a scene */
   428→  useEffect(() => {
   429→    if (prefersReducedMotion()) return;
   430→    const el = root.current;
   431→    if (!el) return;
   432→    const ctx = gsap.context(() => {
   433→      gsap.fromTo(
   434→        el,
   435→        { clipPath: "inset(0 100% 0 0)" },
   436→        {
   437→          clipPath: "inset(0 0% 0 0)",
   438→          duration: 1.4,
   439→          ease: "power3.inOut",
   440→          scrollTrigger: { trigger: el, start: "top 80%", once: true },
   441→          onComplete: () => {
   442→            const o = { v: 97 };
   443→            gsap.to(o, {
   444→              v: 35,
   445→              duration: 2.1,
   446→              ease: "power2.inOut",
   447→              onUpdate: () => apply(o.v),
   448→            });
   449→          },
   450→        },
   451→      );
   452→    });
   453→    return () => ctx.revert();
   454→  }, [apply]);
   455→
   456→  const setFromClientX = useCallback(
   457→    (clientX: number) => {
   458→      const el = root.current;
   459→      if (!el) return;
   460→      const r = el.getBoundingClientRect();
   461→      apply(clamp(((clientX - r.left) / r.width) * 100, 2, 98));
   462→    },
   463→    [apply],
   464→  );
   465→
   466→  const onPointerDown = (e: ReactPointerEvent<HTMLDivElement>) => {
   467→    dragging.current = true;
   468→    (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
   469→    setFromClientX(e.clientX);
   470→  };
   471→  const onPointerMove = (e: ReactPointerEvent<HTMLDivElement>) => {
   472→    if (dragging.current) setFromClientX(e.clientX);
   473→  };
   474→  const endDrag = () => {
   475→    dragging.current = false;
   476→  };
   477→
   478→  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
   479→    const step = e.shiftKey ? 10 : 4;
   480→    if (e.key === "ArrowLeft") {
   481→      apply(clamp(posRef.current - step, 2, 98));
   482→      e.preventDefault();
   483→    } else if (e.key === "ArrowRight") {
   484→      apply(clamp(posRef.current + step, 2, 98));
   485→      e.preventDefault();
   486→    } else if (e.key === "Home") {
   487→      apply(2);
   488→      e.preventDefault();
   489→    } else if (e.key === "End") {
   490→      apply(98);
   491→      e.preventDefault();
   492→    }
   493→  };
   494→
   495→  return (
   496→    <div>
   497→      <div
   498→        ref={root}
   499→        role="slider"
   500→        tabIndex={0}
   501→        aria-valuemin={0}
   502→        aria-valuemax={100}
   503→        aria-valuenow={35}
   504→        aria-label="Comparison between optical and radar imagery"
   505→        aria-valuetext="35% radar"
   506→        onPointerDown={onPointerDown}
   507→        onPointerMove={onPointerMove}
   508→        onPointerUp={endDrag}
   509→        onPointerCancel={endDrag}
   510→        onKeyDown={onKeyDown}
   511→        className="relative aspect-[16/9] w-full cursor-ew-resize touch-pan-y select-none overflow-hidden rounded-md border border-white/15 bg-[#0a0f18] shadow-[0_30px_90px_rgba(0,0,0,0.5)] focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-4 focus-visible:outline-[var(--accent)]"
   512→      >
   513→        {/* optical (base) */}
   514→        <div
   515→          className="absolute inset-0 bg-cover bg-center"
   516→          style={{ backgroundImage: "url(/landscapes/earthquake.webp)" }}
   517→          aria-hidden
   518→        />
   519→        {/* radar (clipped) */}
   520→        <div
   521→          ref={clip}
   522→          className="absolute inset-0 bg-cover bg-center"
   523→          style={{
   524→            backgroundImage: "url(/landscapes/sar.webp)",
   525→            clipPath: "inset(0 0 0 35%)",
   526→          }}
   527→          aria-hidden
   528→        />
   529→
   530→        <span className="pointer-events-none absolute left-3 top-3 rounded-sm bg-[#04070d]/78 px-2.5 py-1.5 text-[10px] font-medium uppercase tracking-[0.2em] text-white/80 backdrop-blur-sm">
   531→          {EVIDENCE.compareLabels.left}
   532→        </span>
   533→        <span className="pointer-events-none absolute right-3 top-3 rounded-sm bg-[#04070d]/78 px-2.5 py-1.5 text-[10px] font-medium uppercase tracking-[0.2em] text-white/80 backdrop-blur-sm">
   534→          {EVIDENCE.compareLabels.right}
   535→        </span>
   536→
   537→        <div
   538→          ref={divider}
   539→          className="pointer-events-none absolute inset-y-0 w-px bg-white/90"
   540→          style={{ left: "35%" }}
   541→        >
   542→          <span className="absolute left-1/2 top-1/2 flex h-10 w-10 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-full border border-white/75 bg-[#04070d]/75 text-white backdrop-blur-sm">
   543→            <MoveHorizontal className="h-4 w-4" aria-hidden />
   544→          </span>

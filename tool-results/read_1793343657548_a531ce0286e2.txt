   200→   200→

   203→   203→      const B1 = 0.335;
   204→   204→      const B2 = 0.67;
   205→   205→
   206→   206→      /* initial states — everything waits in the dark */
   207→   207→      gsap.set([l1, l2], { autoAlpha: 0 });
   208→   208→      gsap.set([t0, t1, t2], { autoAlpha: 0, y: 48 });
   209→   209→      gsap.set(l0, { scale: 1.16, autoAlpha: 0 });

   295→   295→      ref={root}
   296→   296→      id="systems"
   297→   297→      data-frame
   298→   298→      role="region"
   299→   299→      aria-label="The changing systems"

   305→   305→        className="sticky top-0 h-svh overflow-hidden bg-[#04070d]"
   306→   306→        style={{ ["--accent" as string]: SECTORS[0].theme.accent }}

   404→   404→/* the compare slider — optical vs radar, draggable                     */
   405→   405→/* ------------------------------------------------------------------ */
   406→   406→
   407→   407→function CompareSlider() {
   408→   408→  const root = useRef<HTMLDivElement>(null);
   409→   409→  const clip = useRef<HTMLDivElement>(null);
   410→   410→  const divider = useRef<HTMLDivElement>(null);
   411→   411→  const readout = useRef<HTMLSpanElement>(null);
   412→   412→  const posRef = useRef(35);
   413→   413→  const dragging = useRef(false);
   414→   414→
   415→   415→  const apply = useCallback((v: number) => {
   416→   416→    posRef.current = v;
   417→   417→    if (divider.current) divider.current.style.left = `${v}%`;
   418→   418→    if (clip.current) clip.current.style.clipPath = `inset(0 0 0 ${v}%)`;
   419→   419→    if (readout.current)
   420→   420→      readout.current.textContent = `RADAR ${Math.round(v)}%`;
   421→   421→    if (root.current)
   422→   422→      root.current.setAttribute("aria-valuenow", String(Math.round(v)));
   423→   423→  }, []);
   424→   424→
   425→   425→  /* the radar intro — the frame wipes in, then the divider makes one
   426→   426→     scanning pass across the scene before settling, the way the engine
   427→   427→     itself sweeps a scene */
   428→   428→  useEffect(() => {
   429→   429→    if (prefersReducedMotion()) return;
   430→   430→    const el = root.current;
   431→   431→    if (!el) return;
   432→   432→    const ctx = gsap.context(() => {
   433→   433→      gsap.fromTo(
   434→   434→        el,
   435→   435→        { clipPath: "inset(0 100% 0 0)" },
   436→   436→        {
   437→   437→          clipPath: "inset(0 0% 0 0)",
   438→   438→          duration: 1.4,
   439→   439→          ease: "power3.inOut",
   440→   440→          scrollTrigger: { trigger: el, start: "top 80%", once: true },
   441→   441→          onComplete: () => {
   442→   442→            const o = { v: 97 };
   443→   443→            gsap.to(o, {
   444→   444→              v: 35,
   445→   445→              duration: 2.1,
   446→   446→              ease: "power2.inOut",
   447→   447→              onUpdate: () => apply(o.v),
   448→   448→            });
   449→   449→          },
   450→   450→        },
   451→   451→      );
   452→   452→    });
   453→   453→    return () => ctx.revert();
   454→   454→  }, [apply]);
   455→   455→
   456→   456→  const setFromClientX = useCallback(
   457→   457→    (clientX: number) => {
   458→   458→      const el = root.current;
   459→   459→      if (!el) return;
   460→   460→      const r = el.getBoundingClientRect();
   461→   461→      apply(clamp(((clientX - r.left) / r.width) * 100, 2, 98));
   462→   462→    },
   463→   463→    [apply],
